cmake_minimum_required(VERSION 3.20)
project(ArctraWinBackend LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(arctrawin_snapshot snapshot.cpp vendor/arctra/src/cpf.cpp vendor/arctra/src/coordinates.cpp)
target_include_directories(arctrawin_snapshot PUBLIC . vendor/arctra/include)
target_compile_features(arctrawin_snapshot PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(arctrawin_snapshot PRIVATE /W4 /permissive- /EHsc)
endif()
add_executable(arctrawin-export main.cpp)
target_link_libraries(arctrawin-export PRIVATE arctrawin_snapshot)
add_library(arctrawin_live live_worker.cpp)
target_include_directories(arctrawin_live PUBLIC vendor)
target_link_libraries(arctrawin_live PUBLIC arctrawin_snapshot)
add_executable(arctrawin-live live_main.cpp)
target_link_libraries(arctrawin-live PRIVATE arctrawin_live)
include(CTest)
if(BUILD_TESTING)
  add_executable(snapshot-tests ../../tests/Backend.Tests/snapshot_tests.cpp)
  target_link_libraries(snapshot-tests PRIVATE arctrawin_snapshot)
  add_test(NAME snapshot-boundaries COMMAND snapshot-tests "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/Backend.Tests/fixtures/lageos1_cpf_261110_excerpt.dgf")
  add_executable(live-tests ../../tests/Backend.Tests/live_tests.cpp)
  target_link_libraries(live-tests PRIVATE arctrawin_live)
  add_test(NAME live-coverage COMMAND live-tests "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/Backend.Tests/fixtures/lageos1_cpf_261110_excerpt.dgf")
endif()
